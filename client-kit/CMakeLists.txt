cmake_minimum_required(VERSION 3.20)
project(BeetleRemoteClient LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "Windows x64 is required")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(beetle-remote client.cpp remote.cpp)
target_include_directories(beetle-remote PRIVATE include SDL2/include)
target_compile_definitions(beetle-remote PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN _CRT_SECURE_NO_WARNINGS)
target_link_libraries(beetle-remote PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/lib/SDL2.lib" ws2_32 bcrypt)
add_custom_command(TARGET beetle-remote POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
    "${CMAKE_CURRENT_SOURCE_DIR}/../SDL2.dll" "$<TARGET_FILE_DIR:beetle-remote>")
if(MSVC)
    target_link_options(beetle-remote PRIVATE "$<$<CONFIG:Release>:/SUBSYSTEM:WINDOWS>" "$<$<CONFIG:Release>:/ENTRY:mainCRTStartup>")
endif()
enable_testing()
add_test(NAME remote_loopback COMMAND beetle-remote --self-test)
